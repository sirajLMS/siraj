### 2.1 Replace fragile alpha-suffix string concatenation in `LoginView`
**File:** `frontend/views/auth/login/LoginView.tsx` (lines 16, 21)

```diff
+import { alpha } from "@mui/material/styles";
 const StyledPageContainer = styled(Box)(({ theme }) => ({
   ...
-  background: `linear-gradient(135deg, ${theme.palette.primary.main}22 0%, ...)`,
+  background: `linear-gradient(135deg, ${alpha(theme.palette.primary.main, 0.13)} 0%, ...)`,
   "&::before": {
     ...
```
